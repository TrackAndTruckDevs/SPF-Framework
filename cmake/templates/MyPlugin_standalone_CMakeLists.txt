cmake_minimum_required(VERSION 4.4)

  set(PROJECT_VERSION_TWEAK 0)
endif()

add_compile_definitions(_AMD64_)

# --- C++ Standard ---
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# --- Global MSVC Runtime Configuration (/MT) ---
# This ensures static runtime linking, avoiding "Win32 Error 126" on
# machines without VC++ Redistributables.
# Also replaces /EHsc with /EHa to catch SEH exceptions cleanly.
if(MSVC)
    set(variables
        CMAKE_C_FLAGS CMAKE_C_FLAGS_DEBUG CMAKE_C_FLAGS_RELEASE CMAKE_C_FLAGS_RELWITHDEBINFO CMAKE_C_FLAGS_MINSIZEREL
        CMAKE_CXX_FLAGS CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE CMAKE_CXX_FLAGS_RELWITHDEBINFO CMAKE_CXX_FLAGS_MINSIZEREL
    )
    foreach(variable ${variables})
        if(${variable} MATCHES "/MD")
            string(REPLACE "/MD" "/MT" ${variable} "${${variable}}")
        endif()
        if(${variable} MATCHES "/EHsc")
            string(REPLACE "/EHsc" "/EHa" ${variable} "${${variable}}")
        elseif(${variable} MATCHES "/EHs")
            string(REPLACE "/EHs" "/EHa" ${variable} "${${variable}}")
        endif()
    endforeach()
    add_compile_options(/utf-8)
endif()
# --- End of Global MSVC Runtime Configuration ---

# --- Plugin Library ---

# --- SPF API Headers ---
# The SPF API header files are provided in the SPF_API/ subfolder

    "${CMAKE_CURRENT_SOURCE_DIR}/SPF_API"
)

# --- MinGW Static Linking ---
# Ensures static runtime libraries for MinGW cross-compilation.
if(MINGW)
    set(MINGW_STATIC_FLAGS "-static-libgcc;-static-libstdc++")
    set(MINGW_WINPTHREAD_FLAGS "-Wl,-Bstatic;-lwinpthread;-Wl,-Bdynamic")
    target_link_options(${PLUGIN_NAME} PRIVATE
        ${MINGW_STATIC_FLAGS}
        ${MINGW_WINPTHREAD_FLAGS}
    )

    # MinSizeRel uses -Os, which suppresses inlining of libstdc++'s
    # always_inline std::string move-constructors, forcing external
    # references to std::__cxx11::basic_string(&&) that the static
    # libstdc++.a does not provide -> linker 'undefined reference'.
    # Use -O2 instead (still small/fast) so those symbols inline away.
    if(CMAKE_BUILD_TYPE STREQUAL "MinSizeRel")
        string(REPLACE "-Os" "-O2" CMAKE_CXX_FLAGS_MINSIZEREL "${CMAKE_CXX_FLAGS_MINSIZEREL}")
        string(REPLACE "-Os" "-O2" CMAKE_C_FLAGS_MINSIZEREL "${CMAKE_C_FLAGS_MINSIZEREL}")
    endif()
endif()
# --- End of MinGW Static Linking ---

# --- Code Signing ---
# Set SPF_SIGN_CERT to your .pfx path in CMakeUserPresets.json
# Linux/WSL: osslsigncode is required: sudo apt install osslsigncode
# Windows:   signtool is required — install Windows SDK (Visual Studio Installer → Individual Components → Windows 10/11 SDK)
set(SPF_SIGN_CERT "" CACHE FILEPATH "Path to .pfx certificate (set in CMakeUserPresets.json)")
set(SPF_SIGN_PASS "" CACHE STRING "Certificate password (empty if none)")

message(STATUS "  [SIGN]     ${SPF_SIGN_CERT}")

if(SPF_SIGN_CERT AND EXISTS "${SPF_SIGN_CERT}")
    # Try osslsigncode first (works on Linux, WSL, macOS, and Windows if installed)
    find_program(OSLSIGNCODE_EXECUTABLE osslsigncode)
    if(OSLSIGNCODE_EXECUTABLE)
        message(STATUS "           ✅ signing enabled (osslsigncode)")
        set(_SIGN_ARGS sign -pkcs12 "${SPF_SIGN_CERT}")
        if(SPF_SIGN_PASS)
            list(APPEND _SIGN_ARGS -pass "${SPF_SIGN_PASS}")
        endif()
        list(APPEND _SIGN_ARGS
            -in "$<TARGET_FILE:${PLUGIN_NAME}>"
            -out "$<TARGET_FILE:${PLUGIN_NAME}>_signed"
        )
        add_custom_command(TARGET ${PLUGIN_NAME} POST_BUILD
            COMMAND "${OSLSIGNCODE_EXECUTABLE}" ${_SIGN_ARGS}
            COMMAND ${CMAKE_COMMAND} -E rename
                "$<TARGET_FILE:${PLUGIN_NAME}>_signed"
                "$<TARGET_FILE:${PLUGIN_NAME}>"
            COMMENT "  ✅ signing: ${PLUGIN_NAME}.dll (Track'n'Truck Devs)"
        )
    else()
        # Fallback: try signtool from Windows SDK
        find_program(SIGNTOOL_EXECUTABLE signtool
            HINTS
                "$ENV{WindowsSdkDir}/bin/$ENV{WindowsSDKVersion}/x64"
                "C:/Program Files (x86)/Windows Kits/10/bin/10.0.22621.0/x64"
                "C:/Program Files (x86)/Windows Kits/10/bin/10.0.26100.0/x64"
        )
        if(SIGNTOOL_EXECUTABLE)
            message(STATUS "           ✅ signing enabled (signtool)")
            set(_SIGN_CMD "${SIGNTOOL_EXECUTABLE}" sign
                /f "${SPF_SIGN_CERT}"
                /td sha256 /fd sha256
            )
            if(SPF_SIGN_PASS)
                list(APPEND _SIGN_CMD /p "${SPF_SIGN_PASS}")
            endif()
            list(APPEND _SIGN_CMD "$<TARGET_FILE:${PLUGIN_NAME}>")
            add_custom_command(TARGET ${PLUGIN_NAME} POST_BUILD
                COMMAND ${_SIGN_CMD}
                COMMENT "  ✅ signing: ${PLUGIN_NAME}.dll (Track'n'Truck Devs)"
            )
        else()
            message(WARNING "  ⚠️  no signing tool found (install osslsigncode or signtool) — DLL will NOT be signed")
        endif()
    endif()
else()
    message(STATUS "           ⚠️  no cert set — signing skipped")
endif()

        COMMAND ${CMAKE_COMMAND} -E make_directory "${DEPLOY_DIR}"

    )

# --- clangd compile_commands.json (POST_BUILD) ---
# clangd (via .clangd's CompilationDatabase: build) looks for
# build/compile_commands.json. When using CMake presets the file
# ends up in build/<preset>/compile_commands.json instead.
# This POST_BUILD step copies it so clangd can find it.
if(CMAKE_EXPORT_COMPILE_COMMANDS AND NOT CMAKE_GENERATOR MATCHES "Visual Studio")
    string(COMPARE NOTEQUAL "${CMAKE_BINARY_DIR}" "${CMAKE_SOURCE_DIR}/build" NON_DEFAULT_BUILD)
    if(NON_DEFAULT_BUILD)
        set(CLANGD_LINK "${CMAKE_SOURCE_DIR}/compile_commands.json")
        message(STATUS "  clangd:    synced after build → ${CLANGD_LINK}")
        add_custom_command(TARGET ${PLUGIN_NAME} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${CMAKE_BINARY_DIR}/compile_commands.json"
                "${CLANGD_LINK}"
            COMMENT "  clangd:    synced compile_commands.json"
        )
    endif()
endif()
